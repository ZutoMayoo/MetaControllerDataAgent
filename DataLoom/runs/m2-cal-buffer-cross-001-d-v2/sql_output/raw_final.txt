  FROM "Booking" b

),
windows AS (
  SELECT 'BENCH-BUF-OK' AS label, TIMESTAMP '2026-11-03 09:00:00' AS ws, TIMESTAMP '2026-11-03 09:30:00' AS we

  SELECT 'BENCH-BUF-EARLY', TIMESTAMP '2026-11-03 09:30:00', TIMESTAMP '2026-11-03 10:00:00'
